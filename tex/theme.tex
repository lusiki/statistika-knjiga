% Autori poravnati lijevo, jedan ispod drugoga (Jost je širi od prethodnog
% pisma, pa dva stupca lome afilijacije).
\newcommand{\bkcoverauthor}[2]{%
  {\sffamily\large\color{bkink}#1\par}%
  \vskip 0.2em
  {\sffamily\fontsize{7pt}{9pt}\selectfont\color{bkmuted}#2\par}%
}
% Kompozicija naslovnice: tri ugniježđene kupole (tinta, siva, bijela s
% obrubom) nad osnovicom, os kroz sredinu i disk s brojem „101".

    \filldraw[fill=white, draw=bkink, line width=0.8pt] (47,0) arc[start angle=180, end angle=0, radius=13] -- cycle;

    \fill[bkink] (101,80) circle[radius=17];
    \node[text=white, inner sep=0pt, font=\sffamily\bfseries\fontsize{25}{25}\selectfont] at (101,80.3) {101};
    \fill[bkink] (0,-0.85) rectangle (120,0);
  \end{tikzpicture}}
