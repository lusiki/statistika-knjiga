% =====================================================================
%  PEDAGOŠKE KUTIJE  (omata ih pdf-filters/book-callouts.lua)
% =====================================================================
%  Svaka kutija razlikuje se CRTOM I OZNAKOM, ne ispunom u boji:
%
%    Vinjeta            1,2 pt gornja crta + vlas donja, displejni kurziv
%    Statistika u divljini   bijela ploha u 1,6 pt crnom okviru
%    Pitajte model      točkasti okvir, upit u monou na 4 % sivoj
%    Nađite grešku      2,4 pt gornja crta, bez ispune
%    Razrađeni primjer  viseći mono brojevi koraka
%
%  Natpisi se predaju VEĆ VELIKIM SLOVIMA, bez \MakeUppercase. Razlog je Đ:
%  automatsko povećavanje slova ispisalo ga je kao goli D („NADITE GREŠKU"),
%  dok literalno Đ prolazi ispravno.
% =====================================================================
\usepackage[most]{tcolorbox}
\usepackage{marginnote}

\newcommand{\bklbl}[1]{{\eyebrow{#1}\par}\vspace{7pt}}

% Vinjeta
\newenvironment{calloutvinjeta}{%
  \par\vspace{6pt}{\color{bkink}\hrule height 1.2pt}\vspace{11pt}%
  \bklbl{VINJETA}%
  \bkdisplay\itshape\fontsize{13pt}{18pt}\selectfont\color{bkink}\raggedright
}{%
  \par\vspace{11pt}{\color{bkrule}\hrule height 0.4pt}\vspace{16pt}\normalfont\color{bktext}%
}

% Statistika u divljini — na zaslonu obrnuta tamna traka; u tisku bi puna
% crna ploha progutala tekst i papir, pa se obrće u okvir.
\newenvironment{calloutdivljina}{%
  \begin{tcolorbox}[breakable, enhanced, sharp corners, boxrule=1.6pt,
    colframe=bkink, colback=white,
    left=14pt, right=14pt, top=13pt, bottom=13pt,
    before skip=16pt, after skip=16pt]
  \bklbl{STATISTIKA U DIVLJINI}%
  \fontsize{9.5pt}{13.5pt}\selectfont
}{\end{tcolorbox}}

% Pitajte model — točkasti okvir, glas stroja
\newenvironment{calloutmodel}{%
  \begin{tcolorbox}[breakable, enhanced, sharp corners, frame hidden,
    boxrule=0pt, colback=white,
    borderline={0.6pt}{0pt}{bkmuted, dotted},
    left=12pt, right=12pt, top=11pt, bottom=11pt,
    before skip=16pt, after skip=16pt]
  \bklbl{PITAJTE MODEL}%
  \fontsize{9.5pt}{13.5pt}\selectfont
}{\end{tcolorbox}}

% Nađite grešku — teška gornja crta, bez ispune
\newenvironment{calloutgreska}{%
  \par\vspace{14pt}{\color{bkink}\hrule height 2.4pt}\vspace{10pt}%
  \bklbl{NAĐITE GREŠKU}%
  \fontsize{9.5pt}{13.5pt}\selectfont
}{\par\vspace{14pt}\normalfont\normalsize}

% Razrađeni primjer
\newenvironment{calloutprimjer}{%
  \par\vspace{16pt}\bklbl{RAZRAĐENI PRIMJER}%
  \fontsize{9.5pt}{13.5pt}\selectfont
}{\par\vspace{14pt}\normalfont\normalsize}

% Sažetak — uvučena traka
\newenvironment{calloutsazetak}{%
  \begin{tcolorbox}[breakable, enhanced, sharp corners, boxrule=0pt, colback=bkwash,
    left=14pt, right=14pt, top=13pt, bottom=13pt,
    before skip=18pt, after skip=14pt]
  \bklbl{SAŽETAK POGLAVLJA}%
  \fontsize{9.5pt}{13.5pt}\selectfont
}{\end{tcolorbox}}

% Definicija — 1,6 pt lijeva crta + 4 % siva ispuna. Quarto sam ispisuje
% oznaku „Definicija 1", pa je ovdje nema.
\newenvironment{calloutdefinicija}{%
  \begin{tcolorbox}[breakable, enhanced, sharp corners, boxrule=0pt,
    leftrule=1.6pt, colframe=bkink, colback=bkwash,
    left=12pt, right=12pt, top=11pt, bottom=11pt,
    before skip=14pt, after skip=14pt]
  \fontsize{9.5pt}{13.5pt}\selectfont
}{\end{tcolorbox}}
